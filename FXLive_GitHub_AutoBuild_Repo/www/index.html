包含生命周期监听、30秒定时器及主币(CNY)兑罗马尼亚列伊(RON)状态管理

import 'dart:async';
import 'dart:convert';
import 'package:flutter/material.dart';
import 'package:http/http.dart' as http;

class FxRateController extends ChangeNotifier with WidgetsBindingObserver {
  String baseCurrency = 'CNY'; // 主币：默认人民币
  String targetCurrency = 'RON'; // 次币：默认罗马尼亚列伊
  double inputAmount = 1.0;
  Map<String, double> rates = {'CNY': 1.0, 'RON': 0.6385, 'USD': 0.1382, 'EUR': 0.1284};
  int refreshIntervalSeconds = 30; // 支持用户按需调节：30秒 / 1分 / 3分 / 5分 / 最长30分钟(1800秒)
  Timer? _pollingTimer;
  DateTime lastUpdated = DateTime.now();

  FxRateController() {
    WidgetsBinding.instance.addObserver(this);
    fetchLatestRates(); // 1. 启动应用立即更新一次
    _startPolling();    // 2. 按用户设定的省电时间间隔定时刷新
  }

  void setRefreshInterval(int seconds) {
    refreshIntervalSeconds = seconds.clamp(30, 1800);
    _startPolling();
    notifyListeners();
  }

  void _startPolling() {
    _pollingTimer?.cancel();
    _pollingTimer = Timer.periodic(Duration(seconds: refreshIntervalSeconds), (_) {
      fetchLatestRates();
    });
  }

  // 监听手机从后台切回前台：立即刷新并重启30秒计时
  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    if (state == AppLifecycleState.resumed) {
      fetchLatestRates();
      _startPolling();
    } else if (state == AppLifecycleState.paused) {
      _pollingTimer?.cancel(); // 后台省电
    }
  }

  Future<void> fetchLatestRates() async {
    try {
      final uri = Uri.parse('https://open.er-api.com/v6/latest/$baseCurrency');
      final res = await http.get(uri).timeout(const Duration(seconds: 6));
      if (res.statusCode == 200) {
        final data = jsonDecode(res.body);
        final Map<String, dynamic> rawRates = data['rates'];
        rates = rawRates.map((k, v) => MapEntry(k, (v as num).toDouble()));
        lastUpdated = DateTime.now();
        notifyListeners();
      }
    } catch (e) {
      debugPrint('使用本地缓存汇率兜底: $e');
    }
  }

  // 按当时最新费率实时计算兑换结果
  double convert(String targetCode) {
    final rate = rates[targetCode] ?? 0.0;
    return inputAmount * rate;
  }
}
